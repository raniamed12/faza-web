import fs from "node:fs";
const gj = JSON.parse(fs.readFileSync(process.argv[2], "utf8"));
const SLUG = {"Tunis":"tunis","Ariana":"ariana","Ben Arous":"ben-arous","Manouba":"manouba","Nabeul":"nabeul","Zaghouan":"zaghouan",
"Bizerte":"bizerte","Béja":"beja","Jendouba":"jendouba","El Kef":"le-kef","Siliana":"siliana","Sousse":"sousse","Monastir":"monastir",
"Mahdia":"mahdia","Sfax":"sfax","Kairouan":"kairouan","Kasserine":"kasserine","Sidi Bouzid":"sidi-bouzid","Gabès":"gabes",
"Médenine":"medenine","Tataouine":"tataouine","Gafsa":"gafsa","Tozeur":"tozeur","Kébili":"kebili"};
const rad = Math.PI / 180;
const mx = (lon) => lon * rad;
const my = (lat) => Math.log(Math.tan(Math.PI / 4 + (lat * rad) / 2));
// bounds
let minX = 1e9, maxX = -1e9, minY = 1e9, maxY = -1e9;
const polysOf = (g) => (g.type === "Polygon" ? [g.coordinates] : g.coordinates);
for (const f of gj.features) for (const poly of polysOf(f.geometry)) for (const ring of poly) for (const [lo, la] of ring) {
  const x = mx(lo), y = my(la);
  minX = Math.min(minX, x); maxX = Math.max(maxX, x); minY = Math.min(minY, y); maxY = Math.max(maxY, y);
}
const S = 560 / (maxY - minY); // height 560
const pad = 14;
const W = Math.round((maxX - minX) * S + pad * 2), H = Math.round((maxY - minY) * S + pad * 2);
const P = ([lo, la]) => [(mx(lo) - minX) * S + pad, (maxY - my(la)) * S + pad];
const r1 = (n) => Math.round(n * 10) / 10;
const regions = [];
for (const f of gj.features) {
  const slug = SLUG[f.properties.shapeName];
  if (!slug) throw new Error("unmapped " + f.properties.shapeName);
  let d = "", best = null, bestA = -1;
  for (const poly of polysOf(f.geometry)) {
    poly.forEach((ring, ri) => {
      let last = "", out = [];
      for (const c of ring) { const [x, y] = P(c); const s = r1(x) + " " + r1(y); if (s !== last) { out.push(s); last = s; } }
      d += "M" + out.join("L") + "Z";
      if (ri === 0) {
        let a = 0, cx = 0, cy = 0; const pts = ring.map(P);
        for (let i = 0; i < pts.length - 1; i++) { const [x0, y0] = pts[i], [x1, y1] = pts[i + 1]; const k = x0 * y1 - x1 * y0; a += k; cx += (x0 + x1) * k; cy += (y0 + y1) * k; }
        a /= 2; if (Math.abs(a) > bestA) { bestA = Math.abs(a); best = [cx / (6 * a), cy / (6 * a)]; }
      }
    });
  }
  regions.push({ slug, name: f.properties.shapeName, d, cx: r1(best[0]), cy: r1(best[1]) });
}
regions.sort((a, b) => a.slug.localeCompare(b.slug));
const out = `// Fichier généré automatiquement à partir de geoBoundaries (CC BY 4.0, Runfola et al., 2020).\n// Projection Mercator, ${regions.length} gouvernorats. Ne pas modifier à la main.\nexport const MAP_VIEWBOX = { width: ${W}, height: ${H} };\nexport const MAP_REGIONS = ${JSON.stringify(regions)};\n`;
fs.writeFileSync(process.argv[3], out);
console.log("viewBox", W, H, "regions", regions.length, "bytes", out.length);
